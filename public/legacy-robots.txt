# robots.txt for the legacy Netlify host while the domain move completes.
#
# This file is served with HTTP 200 on purpose. Search Console's
# change-of-address check reads robots.txt on the OLD host; a robots.txt that
# redirects to another host is treated as a block, which makes Googlebot unable
# to fetch the old pages at all and fails the whole validation with
# "Couldn't fetch the page".
#
# It allows everything, because the old host does nothing now except answer 301
# redirects to https://freluxtools.com/ - and Google must be able to read those.
User-agent: *
Allow: /
Allow: /ads.txt
Allow: /app-ads.txt

User-agent: Google-adstxt
Allow: /ads.txt
Allow: /app-ads.txt

User-agent: Mediapartners-Google
Allow: /
Allow: /ads.txt
Allow: /app-ads.txt
